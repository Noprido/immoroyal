<script>
// ─── État global du formulaire ────────────────────────────────────
let estVente = false;
let estTerrain = false;

// ─── Toggle transaction ───────────────────────────────────────────
function onTransactionChange(radio) {
  estVente = radio.value === 'vente';
  updateFinancesUI();
}

function onTypeBienChange(select) {
  estTerrain = select.value === 'Terrain';

  // Auto-sélectionner vente si type avec vente par défaut
  const opt = select.options[select.selectedIndex];
  if (opt.dataset.venteDefaut === 'true') {
    document.querySelector('input[name="typeTransaction"][value="vente"]').checked = true;
    estVente = true;
  }

  updateFinancesUI();
  updateCaracteristiquesUI();
}

function updateFinancesUI() {
  const groupDuree = document.getElementById('groupDuree');
  const labelPrix = document.getElementById('labelPrix');
  const inputLoyer = document.getElementById('inputLoyer');
  const inputPrix = document.getElementById('inputPrix');
  const champsLocation = document.querySelectorAll('.form-location-only');

  const groupCommission = document.getElementById('groupCommission');


  if (estVente) {

    groupCommission.style.display = 'none';
    groupCommission.querySelector('input').removeAttribute('required');
    groupCommission.querySelector('input').value = '0';

    labelPrix.textContent = 'Prix de vente *';
    inputLoyer.style.display = 'none';
    inputLoyer.removeAttribute('required');
    inputLoyer.value = '';
    inputPrix.style.display = 'block';
    inputPrix.setAttribute('required', 'required');
    groupDuree.style.display = 'none';
    groupDuree.querySelector('select').removeAttribute('required');
    champsLocation.forEach(el => {
      el.style.display = 'none';
      el.querySelector('input').removeAttribute('required');
      el.querySelector('input').value = '0';
    });
  } else {
    groupCommission.style.display = 'flex';
    groupCommission.querySelector('input').setAttribute('required', 'required');

    labelPrix.textContent = 'Loyer *';
    inputLoyer.style.display = 'block';
    inputLoyer.setAttribute('required', 'required');
    inputPrix.style.display = 'none';
    inputPrix.removeAttribute('required');
    inputPrix.value = '';
    groupDuree.style.display = 'flex';
    groupDuree.querySelector('select').setAttribute('required', 'required');
    champsLocation.forEach(el => {
      el.style.display = 'flex';
      el.querySelector('input').setAttribute('required', 'required');
    });
  }
}

const CATEGORIES = <%- JSON.stringify(categories) %>;

function getCategorie(typeBien) {
  return CATEGORIES.find(c => c.value === typeBien) || null;
}
function champsPourType(typeBien) {
  const cat = getCategorie(typeBien);
  return cat ? cat.champsCaracteristiques : [];
}
function afficheDimensionPourType(typeBien) {
  const cat = getCategorie(typeBien);
  return cat ? cat.afficheDimension : false;
}

function updateCaracteristiquesUI() {
  const typeBien = document.getElementById('typeBien').value;
  const champs = champsPourType(typeBien);
  const dimensionActive = afficheDimensionPourType(typeBien);
  estTerrain = dimensionActive;

  const cautionEau  = document.getElementById('groupCautionEau');
  const cautionElec = document.getElementById('groupCautionElec');
  [cautionEau, cautionElec].forEach(el => {
    if (!el) return;
    el.style.display = dimensionActive ? 'none' : 'flex';
    const input = el.querySelector('input');
    if (dimensionActive) { input.removeAttribute('required'); input.value = '0'; }
    else input.setAttribute('required', 'required');
  });

  const cardCarac     = document.getElementById('cardCaracteristiques');
  const cardTerrain   = document.getElementById('cardTerrain');
  const inputDim      = document.getElementById('inputDimension');
  const groupStanding = document.getElementById('groupStanding');

  if (dimensionActive) {
    cardCarac.style.display = 'none';
    cardTerrain.style.display = 'block';
    cardCarac.querySelectorAll('input, select').forEach(el => el.removeAttribute('required'));
    if (inputDim) inputDim.setAttribute('required', 'required');
  } else {
    cardCarac.style.display = 'block';
    cardTerrain.style.display = 'none';
    if (inputDim) inputDim.removeAttribute('required');

    document.querySelectorAll('.field-carac').forEach(group => {
      const champ = group.dataset.field;
      const actif = champs.includes(champ);
      group.style.display = actif ? 'block' : 'none';
      const input = group.querySelector('input');
      if (actif) input.setAttribute('required', 'required');
      else { input.removeAttribute('required'); input.value = ''; }
    });

    const supportsStanding = champs.includes('standing');
    groupStanding.style.display = supportsStanding ? 'block' : 'none';
    const selStanding = document.getElementById('selectStanding');
    if (supportsStanding) selStanding.setAttribute('required', 'required');
    else { selStanding.removeAttribute('required'); selStanding.value = ''; }
  }

  updateNbPiecesDisplay();
}

function updateNbPiecesDisplay() {
  const total = ['nbChambres', 'nbSalons', 'nbDouches', 'nbCuisines']
    .reduce((sum, id) => sum + (parseInt(document.getElementById(id)?.value) || 0), 0);
  const el = document.getElementById('nbPieces');
  if (el) el.value = total;
}

function pieceWord(n, singulier, pluriel, feminin) {
  n = parseInt(n) || 0;
  if (n === 1) return (feminin ? 'une ' : 'un ') + singulier;
  return n + ' ' + pluriel;
}

function genererTitre() {
  const typeBien = document.getElementById('typeBien').value;
  const villeInput = document.querySelector('[name="ville"]');
  const ville = villeInput ? villeInput.value.trim() : '';
  const cat = getCategorie(typeBien);
  if (!cat || !ville) return '';

  const champs = cat.champsCaracteristiques;
  const supportsStanding = champs.includes('standing');
  const standingEl = document.getElementById('selectStanding');
  const standing = supportsStanding && standingEl ? standingEl.value : '';
  const standingSuffix = (supportsStanding && standing) ? ' ' + standing : '';

  const nbChambres = parseInt(document.getElementById('nbChambres')?.value) || 0;
  const nbSalons   = parseInt(document.getElementById('nbSalons')?.value) || 0;
  const nbDouches  = parseInt(document.getElementById('nbDouches')?.value) || 0;
  const nbCuisines = parseInt(document.getElementById('nbCuisines')?.value) || 0;
  const nbPieces   = nbChambres + nbSalons + nbDouches + nbCuisines;
  const dimension  = parseFloat(document.getElementById('inputDimension')?.value) || 0;

  switch (cat.titreFormat) {
    case 'chambresSalon':
      return pieceWord(nbChambres, 'chambre', 'chambres', true) + ' ' +
             pieceWord(nbSalons, 'salon', 'salons', false) + standingSuffix + ' à ' + ville;
    case 'pieces':
      return typeBien + ' ' + pieceWord(nbPieces, 'pièce', 'pièces', true) + standingSuffix + ' à ' + ville;
    case 'terrain':
      return 'Terrain ' + dimension + 'm² à ' + ville;
    case 'simple':
    default:
      return typeBien + ' à ' + ville;
  }
}

function updateTitrePreview() {
  updateNbPiecesDisplay();
  const titre = genererTitre();
  const preview = document.getElementById('titrePreview');
  if (preview) preview.textContent = titre || '—';
  const hidden = document.getElementById('titreHidden');
  if (hidden) hidden.value = titre;
}

document.addEventListener('DOMContentLoaded', () => {
  const villeInput = document.querySelector('[name="ville"]');
  if (villeInput) villeInput.addEventListener('input', updateTitrePreview);
  updateTitrePreview();
});

document.getElementById('annonceForm')?.addEventListener('submit', (e) => {
  updateTitrePreview();
  const hidden = document.getElementById('titreHidden');
  if (!hidden.value) {
    e.preventDefault();
    alert('Impossible de générer le titre : vérifiez le type de bien et la ville.');
  }
});

// ─── Prévisualisation médias ──────────────────────────────────────
let mediaFiles = [];

function previewMedias(input) {
  const newFiles = Array.from(input.files);
  mediaFiles = [...mediaFiles, ...newFiles];
  renderMediasPreview();
}

function renderMediasPreview() {
  const preview = document.getElementById('mediasPreview');
  const counter = document.getElementById('mediaCounter');
  const counterPhotos = document.getElementById('counterPhotos');
  const counterVideos = document.getElementById('counterVideos');

  let nbPhotos = 0, nbVideos = 0;
  preview.innerHTML = '';

  if (mediaFiles.length === 0) {
    counter.style.display = 'none';
    return;
  }

  counter.style.display = 'block';

  mediaFiles.forEach((file, index) => {
    const isVideo = file.type.startsWith('video/');
    if (isVideo) nbVideos++; else nbPhotos++;

    const div = document.createElement('div');
    div.className = 'media-preview-item';
    div.dataset.index = index;

    if (isVideo) {
      const video = document.createElement('video');
      video.src = URL.createObjectURL(file);
      video.muted = true;
      video.style.cssText = 'width:100%; height:100%; object-fit:cover; border-radius:8px;';
      div.appendChild(video);

      const badge = document.createElement('span');
      badge.className = 'media-badge-video';
      badge.innerHTML = "<i class='bx bx-play-circle'></i> Vidéo";
      div.appendChild(badge);
    } else {
      const img = document.createElement('img');
      img.style.cssText = 'width:100%; height:100%; object-fit:cover; border-radius:8px;';
      const reader = new FileReader();
      reader.onload = e => img.src = e.target.result;
      reader.readAsDataURL(file);
      div.appendChild(img);
    }

    // Bouton supprimer
    const btnDel = document.createElement('button');
    btnDel.type = 'button';
    btnDel.className = 'media-preview-delete';
    btnDel.innerHTML = "<i class='bx bx-x'></i>";
    btnDel.onclick = () => {
      mediaFiles.splice(index, 1);
      renderMediasPreview();
      rebuildFileInput();
    };
    div.appendChild(btnDel);
    preview.appendChild(div);
  });

  counterPhotos.textContent = `${nbPhotos} photo${nbPhotos !== 1 ? 's' : ''}`;
  counterVideos.textContent = `${nbVideos} vidéo${nbVideos !== 1 ? 's' : ''}`;

  // Validation visuelle
  const total = nbPhotos + nbVideos;
  counter.style.color = (total > 10 || nbVideos > 3) ? '#dc3545' : 'var(--grey)';
}

function rebuildFileInput() {
  // Reconstruire le FileList depuis mediaFiles via DataTransfer
  const dt = new DataTransfer();
  mediaFiles.forEach(f => dt.items.add(f));
  document.getElementById('mediaInput').files = dt.files;
}

// Drag & drop sur la zone upload
document.addEventListener('DOMContentLoaded', () => {
  const zone = document.getElementById('uploadZone');
  if (!zone) return;

  zone.addEventListener('dragover', e => {
    e.preventDefault();
    zone.classList.add('drag-over');
  });
  zone.addEventListener('dragleave', () => zone.classList.remove('drag-over'));
  zone.addEventListener('drop', e => {
    e.preventDefault();
    zone.classList.remove('drag-over');
    const files = Array.from(e.dataTransfer.files);
    mediaFiles = [...mediaFiles, ...files];
    renderMediasPreview();
    rebuildFileInput();
  });
});
